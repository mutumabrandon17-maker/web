<script>
  // Basic tab switching for login/register
  const authTabs = document.querySelectorAll(".auth-tab");
  const loginForm = document.getElementById("loginForm");
  const registerForm = document.getElementById("registerForm");
  const authMessage = document.getElementById("authMessage");
  const registerMessage = document.getElementById("registerMessage");
  const authScreen = document.getElementById("authScreen");
  const dashboardContainer = document.getElementById("dashboardContainer");

  authTabs.forEach(tab => {
    tab.addEventListener("click", () => {
      authTabs.forEach(t => t.classList.remove("active"));
      tab.classList.add("active");
      if (tab.dataset.tab === "login") {
        loginForm.style.display = "flex";
        registerForm.style.display = "none";
        authMessage.textContent = "";
      } else {
        loginForm.style.display = "none";
        registerForm.style.display = "flex";
        registerMessage.textContent = "";
      }
    });
  });

  // Users data stored in localStorage for demo
  let users = JSON.parse(localStorage.getItem("mutariiUsers") || "null") || [
    { fullName: "Admin User", username: "admin", password: "1234", biometric: "admin" }
  ];

  function saveUsers() {
    localStorage.setItem("mutariiUsers", JSON.stringify(users));
  }

  // Face API setup
  const video = document.getElementById("video");
  const scanCanvas = document.getElementById("scanCanvas");
  const scanResult = document.getElementById("scanResult");
  const startCameraBtn = document.getElementById("startCameraBtn");
  const stopCameraBtn = document.getElementById("stopCameraBtn");
  const scanFaceBtn = document.getElementById("scanFaceBtn");

  let stream = null;
  let modelsLoaded = false;
  let lastScan = { matched: false, name: null, confidence: 0 };

  async function loadModels() {
    const modelBase = "https://cdn.jsdelivr.net/npm/@vladmandic/face-api/model/";
    await Promise.all([
      faceapi.nets.tinyFaceDetector.loadFromUri(modelBase),
      faceapi.nets.faceLandmark68Net.loadFromUri(modelBase),
      faceapi.nets.faceRecognitionNet.loadFromUri(modelBase)
    ]);
    modelsLoaded = true;
    scanResult.textContent = "AI models loaded. Start camera.";
  }
  loadModels();

  async function startCamera() {
    if (stream) return;
    try {
      stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "user" } });
      video.srcObject = stream;
      await video.play();
      scanCanvas.width = video.videoWidth;
      scanCanvas.height = video.videoHeight;
      scanResult.textContent = "Camera started. Ready to scan.";
    } catch (e) {
      scanResult.textContent = "Camera access denied or unavailable.";
    }
  }

  function stopCamera() {
    if (stream) {
      stream.getTracks().forEach(track => track.stop());
      stream = null;
    }
    video.srcObject = null;
    const ctx = scanCanvas.getContext("2d");
    ctx.clearRect(0, 0, scanCanvas.width, scanCanvas.height);
    scanResult.textContent = "Camera stopped.";
  }

  function drawBox(detection) {
    const ctx = scanCanvas.getContext("2d");
    ctx.clearRect(0, 0, scanCanvas.width, scanCanvas.height);
    if (!detection) return;
    const dims = faceapi.matchDimensions(scanCanvas, video, true);
    const resized = faceapi.resizeResults(detection, dims);
    faceapi.draw.drawDetections(scanCanvas, resized);
  }

  // Fake biometric matching by username or biometric label
  function fakeMatch(usernameInput) {
    const input = (usernameInput || "").toLowerCase().trim();
    const user = users.find(u => u.username.toLowerCase() === input || (u.biometric || "").toLowerCase() === input);
    if (user) {
      return { matched: true, name: user.fullName, confidence: 85 };
    }
    return { matched: false, name: "Unknown face", confidence: 30 };
  }

  async function runFaceScan() {
    if (!modelsLoaded) {
      scanResult.textContent = "Models not loaded yet.";
      return;
    }
    if (!stream) {
      scanResult.textContent = "Start camera first.";
      return;
    }
    scanResult.textContent = "Scanning face...";
    const detection = await faceapi.detectSingleFace(video, new faceapi.TinyFaceDetectorOptions()).withFaceLandmarks().withFaceDescriptor();
    drawBox(detection);
    if (!detection) {
      lastScan = { matched: false, name: null, confidence: 0 };
      scanResult.textContent = "No face detected.";
      return;
    }
    const result = fakeMatch(document.getElementById("loginUsername").value);
    lastScan = result;
    scanResult.textContent = result.matched ? `Matched: ${result.name} (${result.confidence}%)` : "Face not recognized";
  }

  startCameraBtn.onclick = startCamera;
  stopCameraBtn.onclick = stopCamera;
  scanFaceBtn.onclick = runFaceScan;

  // Register form submit
  registerForm.addEventListener("submit", e => {
    e.preventDefault();
    const fullName = document.getElementById("regName").value.trim();
    const username = document.getElementById("regUsername").value.trim();
    const password = document.getElementById("regPassword").value.trim();
    const biometric = document.getElementById("regBiometric").value.trim() || username;

    if (!fullName || !username || !password) {
      registerMessage.textContent = "Please fill all required fields.";
      registerMessage.style.color = "#ff6978";
      return;
    }
    if (users.some(u => u.username.toLowerCase() === username.toLowerCase())) {
      registerMessage.textContent = "Username already exists.";
      registerMessage.style.color = "#ff6978";
      return;
    }
    users.push({ fullName, username, password, biometric });
    saveUsers();
    registerMessage.style.color = "#39e58c";
    registerMessage.textContent = `User ${fullName} registered successfully. You can now login.`;
    registerForm.reset();
  });

  // Login form submit - ONLY username/password required now, face scan NOT mandatory
  loginForm.addEventListener("submit", e => {
    e.preventDefault();
    authMessage.textContent = "";
    const username = document.getElementById("loginUsername").value.trim();
    const password = document.getElementById("loginPassword").value.trim();

    if (!username || !password) {
      authMessage.textContent = "Please enter username and password.";
      return;
    }
    const user = users.find(u => u.username.toLowerCase() === username.toLowerCase() && u.password === password);
    if (!user) {
      authMessage.textContent = "Invalid username or password.";
      return;
    }
    // FACE SCAN CHECK REMOVED - login allowed without face scan
    injectDashboard(user);
  });

  // Inject dashboard HTML dynamically after login
  function injectDashboard(user) {
    authScreen.style.display = "none";
    dashboardContainer.style.display = "flex";
    dashboardContainer.style.flexDirection = "column";
    dashboardContainer.innerHTML = `
      <header class="dash-header">
        <div class="dash-brand">
          <div class="dash-brand-icon">MS</div>
          <div class="dash-brand-text">
            <h2>Mutarii Security System</h2>
            <small>Made in Kenya</small>
          </div>
        </div>
        <div class="dash-header-right">
          <div class="operator-name">Welcome, ${user.fullName}</div>
          <button id="logoutBtn">Logout</button>
        </div>
      </header>
      <nav class="dash-nav">
        <button class="active" data-section="home">Home</button>
        <button data-section="settings">Settings</button>
        <button data-section="cases">Cases</button>
        <button data-section="registration">Registration</button>
        <button data-section="reports">Reports</button>
        <button data-section="mediaUpload">Media Upload</button>
      </nav>
      <section id="home" class="dash-section active">
        <h3>Home</h3>
        <p>Welcome to Mutarii Security System made in Kenya. This is your dashboard home.</p>
      </section>
      <section id="settings" class="dash-section">
        <h3>Settings</h3>
        <p>Configure your system settings here.</p>
      </section>
      <section id="cases" class="dash-section">
        <h3>Cases</h3>
        <p>Manage and view security cases here.</p>
      </section>
      <section id="registration" class="dash-section">
        <h3>Registration</h3>
        <p>Register new operators or personnel here.</p>
      </section>
      <section id="reports" class="dash-section">
        <h3>Reports</h3>
        <p>View system reports and analytics here.</p>
      </section>
      <section id="mediaUpload" class="dash-section" style="flex-direction: column; gap: 15px;">
        <h3>Upload Images or CCTV Footage</h3>

        <div>
          <label for="fileInput">Upload from your device:</label><br />
          <input type="file" id="fileInput" accept="image/*,video/*" />
        </div>

        <div style="margin-top: 15px;">
          <label for="urlInput">Or enter URL of image/video:</label><br />
          <input type="text" id="urlInput" placeholder="https://example.com/video.mp4" style="width: 100%; padding: 8px;" />
          <button id="loadUrlBtn" style="margin-top: 8px; padding: 8px 16px;">Load from URL</button>
        </div>

        <div id="previewArea" style="margin-top: 20px; border: 1px solid #294862; border-radius: 12px; padding: 10px; background: #02070d; min-height: 180px; display: flex; justify-content: center; align-items: center; color: #7087a0;">
          <p>Preview will appear here after upload or URL load.</p>
        </div>

        <button id="runFaceDetectBtn" style="margin-top: 15px; padding: 12px 20px; background: #147fc4; color: white; border: none; border-radius: 8px; cursor: pointer;">
          Run Face Detection
        </button>

        <div id="faceDetectResult" style="margin-top: 15px; color: #39e58c; font-weight: 600;"></div>
      </section>
    `;

    // Setup dashboard tab switching
    const tabs = dashboardContainer.querySelectorAll(".dash-nav button");
    const sections = dashboardContainer.querySelectorAll(".dash-section");
    tabs.forEach(tab => {
      tab.addEventListener("click", () => {
        tabs.forEach(t => t.classList.remove("active"));
        tab.classList.add("active");
        const target = tab.dataset.section;
        sections.forEach(sec => {
          sec.classList.toggle("active", sec.id === target);
        });
      });
    });

    // Logout button
    const logoutBtn = dashboardContainer.querySelector("#logoutBtn");
    logoutBtn.addEventListener("click", () => {
      dashboardContainer.innerHTML = "";
      dashboardContainer.style.display = "none";
      authScreen.style.display = "block";
      lastScan = { matched: false, name: null, confidence: 0 };
      scanResult.textContent = "Waiting for scan...";
      stopCamera();
    });

    // Media Upload elements and logic
    const fileInput = dashboardContainer.querySelector("#fileInput");
    const urlInput = dashboardContainer.querySelector("#urlInput");
    const loadUrlBtn = dashboardContainer.querySelector("#loadUrlBtn");
    const previewArea = dashboardContainer.querySelector("#previewArea");
    const runFaceDetectBtn = dashboardContainer.querySelector("#runFaceDetectBtn");
    const faceDetectResult = dashboardContainer.querySelector("#faceDetectResult");

    let currentMediaElement = null;

    function clearPreview() {
      previewArea.innerHTML = '<p style="color: #7087a0;">Preview will appear here after upload or URL load.</p>';
      faceDetectResult.textContent = '';
      currentMediaElement = null;
    }

    fileInput.addEventListener('change', () => {
      faceDetectResult.textContent = '';
      if (fileInput.files.length === 0) {
        clearPreview();
        return;
      }
      const file = fileInput.files[0];
      const url = URL.createObjectURL(file);
      loadMedia(url, file.type);
    });

    loadUrlBtn.addEventListener('click', () => {
      faceDetectResult.textContent = '';
      const url = urlInput.value.trim();
      if (!url) {
        alert('Please enter a valid URL');
        return;
      }
      try {
        new URL(url);
      } catch {
        alert('Invalid URL format');
        return;
      }
      const ext = url.split('.').pop().toLowerCase();
      let type = '';
      if (['jpg','jpeg','png','gif','bmp','webp'].includes(ext)) {
        type = 'image/*';
      } else if (['mp4','webm','ogg','mov','avi'].includes(ext)) {
        type = 'video/*';
      } else {
        alert('Unsupported media type');
        return;
      }
      loadMedia(url, type);
    });

    function loadMedia(url, type) {
      clearPreview();
      if (type.startsWith('image')) {
        const img = document.createElement('img');
        img.src = url;
        img.style.maxWidth = '100%';
        img.style.borderRadius = '12px';
        img.onload = () => URL.revokeObjectURL(url);
        previewArea.innerHTML = '';
        previewArea.appendChild(img);
        currentMediaElement = img;
      } else if (type.startsWith('video')) {
        const video = document.createElement('video');
        video.src = url;
        video.controls = true;
        video.style.maxWidth = '100%';
        video.style.borderRadius = '12px';
        previewArea.innerHTML = '';
        previewArea.appendChild(video);
        currentMediaElement = video;
      }
    }

    runFaceDetectBtn.addEventListener('click', async () => {
      faceDetectResult.textContent = '';
      if (!currentMediaElement) {
        alert('Please upload or load media first.');
        return;
      }
      faceDetectResult.textContent = 'Running face detection...';

      try {
        if (currentMediaElement.tagName === 'IMG') {
          const detections = await faceapi.detectAllFaces(currentMediaElement, new faceapi.TinyFaceDetectorOptions());
          faceDetectResult.textContent = `Detected ${detections.length} face(s) in the image.`;
        } else if (currentMediaElement.tagName === 'VIDEO') {
          // For video, detect faces on the current frame
          const canvas = faceapi.createCanvasFromMedia(currentMediaElement);
          const displaySize = { width: currentMediaElement.videoWidth, height: currentMediaElement.videoHeight };
          faceapi.matchDimensions(canvas, displaySize);
          const detections = await faceapi.detectAllFaces(currentMediaElement, new faceapi.TinyFaceDetectorOptions());
          faceDetectResult.textContent = `Detected ${detections.length} face(s) in the video frame.`;
        }
      } catch (err) {
        faceDetectResult.textContent = 'Error running face detection.';
        console.error(err);
      }
    });

    stopCamera();
  }
</script>
